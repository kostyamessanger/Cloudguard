// CloudGuard Core 🛡️
// Cookie-based reload protection

(function () {
  function getCount() {
    const m = document.cookie.match(/cloudguard_count=(\d+)/);
    return m ? Number(m[1]) : 0;
  }

  function setCount(v) {
    document.cookie = "cloudguard_count=" + v + "; path=/";
  }

  let count = getCount();
  count++;
  setCount(count);

  if (count >= 5) {
    document.documentElement.innerHTML = `
      
CloudGuard

      
Access blocked

    `;
  }
})();
